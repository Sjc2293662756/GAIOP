## 二、改造方案 → 实施

### 2.1 软件版本：固定 "5.0"，保留手动覆盖入口

#### InspectionFieldMapperService.js — `mapDevice()` line 175

**改动前：**
```js
const softwareVersion = asText(options.softwareVersion || parseSysVersion(aboutHtml) || source.mktVersion || source.uiVersion);
```

**改动后：**
```js
const DEFAULT_SOFTWARE_VERSION = '5.0';
const softwareVersion = asText(options.softwareVersion || DEFAULT_SOFTWARE_VERSION);
```

> `options.softwareVersion` 优先级最高（调用方传了就用调用方的），否则统一用 `"5.0"`。
> 后期如需修改，只需改 `DEFAULT_SOFTWARE_VERSION` 常量。

#### SummaryService.js — `_normalizeDeviceInfo()` line 1339

**改动前：**
```js
softwareVersion: raw.uiVersion || raw.mktVersion || raw.softwareVersion || raw.version || '',
```

**改动后：**
```js

```

### 2.2 序列号：查询保留 + 前缀通用替换

#### InspectionFieldMapperService.js — `formatSerialNumber()` lines 121-128

**改动前：**
```js
function formatSerialNumber(value = '') {
  const text = asText(value);
  if (!text) return '';
  if (/^(ARXVXA|ARX3800)-/i.test(text)) {
    return text.replace(/^(ARXVXA|ARX3800)/i, 'NetInside NAPM 4.0');
  }
  return text;
}
```

**改动后：**
```js
function formatSerialNumber(value = '') {
  const text = asText(value);
  if (!text) return '';
  // 通用前缀替换：将原始前缀（如 ARXVXA、ARX3800）统一替换为 NAPM
  return text.replace(/^[A-Z0-9]+-/i, 'NAPM-');
}
```

#### SummaryService.js — `_normalizeDeviceInfo()` + 新增 `_formatSerialNumber()`

**改动前 (`_normalizeDeviceInfo`)：**
```js
_normalizeDeviceInfo(raw) {
    if (!isPlainObject(raw)) return {};
    return {
      systemName: raw.boxName || raw.systemName || raw.hostname || '',
      ip: raw.address || raw.ip || raw.ipAddress || '',
      softwareVersion: raw.uiVersion || raw.mktVersion || raw.softwareVersion || raw.version || '',
      serialNumber: raw.serialNumber || raw.sn || '',
      uptime: raw.uptime || ''
    };
  }
```

**改动后：**
```js
_normalizeDeviceInfo(raw) {
    if (!isPlainObject(raw)) return {};
    return {
      systemName: raw.boxName || raw.systemName || raw.hostname || '',
      ip: raw.address || raw.ip || raw.ipAddress || '',
      softwareVersion: '5.0',
      serialNumber: this._formatSerialNumber(raw.serialNumber || raw.sn || ''),
      uptime: raw.uptime || ''
    };
  }

  _formatSerialNumber(value = '') {
    const text = String(value || '').trim();
    if (!text) return '';
    // 通用前缀替换：将原始前缀（如 ARXVXA、ARX3800）统一替换为 NAPM
    return text.replace(/^[A-Z0-9]+-/i, 'NAPM-');
  }
```
